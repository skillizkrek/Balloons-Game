    <script>
        const viewport = document.getElementById('phone-viewport');
        const scoreEl = document.getElementById('score');
        const pauseBtn = document.getElementById('pause-btn');
        const waveBtn = document.getElementById('wave-btn');

        let score = 0;
        let isPaused = false;
        let isWaveMode = false;
        let modeChangedOnPause = false;
        let spawnTimeout;

        let BALLOON_WIDTH = 85;
        const HITBOX_EXTRA = 10;
        const MAX_ACTIVE = 3;

        // === РАНДОМ С ЗАЩИТОЙ ОТ НИТКИ ===
        let prevX = -1;

        // === АУДИО ===
        let audioCtx = null;

        function initAudio() {
            if (audioCtx) return;
            try {
                const Ctx = window.AudioContext || window.webkitAudioContext;
                audioCtx = new Ctx();
            } catch (e) {
                audioCtx = null;
            }
        }

        function unlockAudio() {
            initAudio();
            if (audioCtx && audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
        }

        function playClick() {
            if (!audioCtx) return;
            if (audioCtx.state === 'suspended') audioCtx.resume();

            const now = audioCtx.currentTime;

            const noiseLen = Math.floor(audioCtx.sampleRate * 0.008);
            const buf = audioCtx.createBuffer(1, noiseLen, audioCtx.sampleRate);
            const d = buf.getChannelData(0);
            for (let i = 0; i < noiseLen; i++) {
                const t = i / noiseLen;
                d[i] = (Math.random() * 2 - 1) * Math.pow(1 - t, 5);
            }
            const noise = audioCtx.createBufferSource();
            noise.buffer = buf;

            const noiseFilter = audioCtx.createBiquadFilter();
            noiseFilter.type = 'bandpass';
            noiseFilter.frequency.value = 3500;
            noiseFilter.Q.value = 1.0;

            const noiseGain = audioCtx.createGain();
            noiseGain.gain.setValueAtTime(0.5, now);
            noiseGain.gain.exponentialRampToValueAtTime(0.0001, now + 0.01);

            noise.connect(noiseFilter);
            noiseFilter.connect(noiseGain);
            noiseGain.connect(audioCtx.destination);
            noise.start(now);
            noise.stop(now + 0.015);

            const osc1 = audioCtx.createOscillator();
            const gain1 = audioCtx.createGain();
            osc1.type = 'triangle';
            osc1.frequency.setValueAtTime(900, now);
            osc1.frequency.exponentialRampToValueAtTime(500, now + 0.015);
            gain1.gain.setValueAtTime(0.0001, now);
            gain1.gain.exponentialRampToValueAtTime(0.3, now + 0.001);
            gain1.gain.exponentialRampToValueAtTime(0.0001, now + 0.02);
            osc1.connect(gain1);
            gain1.connect(audioCtx.destination);
            osc1.start(now);
            osc1.stop(now + 0.025);

            const osc2 = audioCtx.createOscillator();
            const gain2 = audioCtx.createGain();
            osc2.type = 'sine';
            osc2.frequency.setValueAtTime(1800, now);
            gain2.gain.setValueAtTime(0.0001, now);
            gain2.gain.exponentialRampToValueAtTime(0.08, now + 0.001);
            gain2.gain.exponentialRampToValueAtTime(0.0001, now + 0.012);
            osc2.connect(gain2);
            gain2.connect(audioCtx.destination);
            osc2.start(now);
            osc2.stop(now + 0.015);
        }

        document.addEventListener('touchstart', unlockAudio, { once: true, passive: true });
        document.addEventListener('pointerdown', unlockAudio, { once: true });

        // === Логика ===
        function updateSizes() {
            const W = viewport.clientWidth;
            BALLOON_WIDTH = Math.max(75, Math.min(130, Math.round(W * 0.24)));
        }

        window.addEventListener('resize', updateSizes);
        window.addEventListener('orientationchange', () => setTimeout(updateSizes, 200));
        updateSizes();

        const colors = ['#ff3366', '#33b5e5', '#ffcc00', '#ff6600', '#9933ff', '#33ff33'];
        let lastColorIndex = -1;

        const activeBalloons = new Map();
        let nextBalloonId = 0;

        // Рандом: не даём попасть в тот же X, где был предыдущий шарик
        function pickRandomX() {
            const W = viewport.clientWidth;
            const maxX = W - BALLOON_WIDTH;

            // Первый шарик
            if (prevX < 0) {
                const x = Math.random() * maxX;
                prevX = x;
                return x;
            }

            // 10 попыток найти X далеко от предыдущего
            for (let i = 0; i < 10; i++) {
                const x = Math.random() * maxX;
                if (Math.abs(x - prevX) > BALLOON_WIDTH + 20) {
                    prevX = x;
                    return x;
                }
            }

            // Не нашли — берём что есть
            const fallback = Math.random() * maxX;
            prevX = fallback;
            return fallback;
        }

        function createBalloon(leftPos) {
            const id = nextBalloonId++;

            const container = document.createElement('div');
            const balloon = document.createElement('div');
            const hitbox = document.createElement('div');

            let colorIndex = Math.floor(Math.random() * colors.length);
            while (colorIndex === lastColorIndex) {
                colorIndex = Math.floor(Math.random() * colors.length);
            }
            lastColorIndex = colorIndex;
            const color = colors[colorIndex];

            container.className = 'balloon-container';
            balloon.className = 'balloon';
            hitbox.className = 'balloon-hitbox';

            balloon.style.backgroundColor = color;
            balloon.style.color = color;

            const w = BALLOON_WIDTH;
            const hitboxSize = w + HITBOX_EXTRA * 2;

            balloon.style.width = w + 'px';
            balloon.style.height = w + 'px';

            hitbox.style.width = hitboxSize + 'px';
            hitbox.style.height = hitboxSize + 'px';
            hitbox.style.left = (-HITBOX_EXTRA) + 'px';
            hitbox.style.top = (-HITBOX_EXTRA) + 'px';

            container.style.width = w + 'px';
            container.style.height = (w + 5) + 'px';

            container.appendChild(hitbox);
            container.appendChild(balloon);

            container.style.left = `${leftPos}px`;

            activeBalloons.set(id, { el: container, x: leftPos });

            const duration = isWaveMode ? 1.6 : 6.0;

            let cleanedUp = false;

            const cleanup = () => {
                if (cleanedUp) return;
                cleanedUp = true;
                activeBalloons.delete(id);
                if (container.isConnected) container.remove();
            };

            let popped = false;

            const popHandler = (e) => {
                if (popped) return;
                if (isPaused) return;

                popped = true;

                e.preventDefault();
                e.stopPropagation();

                unlockAudio();
                playClick();

                const r = container.getBoundingClientRect();
                const vr = viewport.getBoundingClientRect();
                const cx = BALLOON_WIDTH / 2;

                for (let i = 0; i < 12; i++) {
                    const p = document.createElement('div');
                    const a = (i / 12) * Math.PI * 2 + Math.random() * 0.5;
                    const d = Math.random() * 60 + 40;
                    p.className = 'particle';
                    p.style.backgroundColor = color;
                    p.style.left = `${r.left - vr.left + cx}px`;
                    p.style.top = `${r.top - vr.top + cx}px`;
                    p.style.setProperty('--dx', `${Math.cos(a) * d}px`);
                    p.style.setProperty('--dy', `${Math.sin(a) * d}px`);
                    viewport.appendChild(p);
                    p.addEventListener('animationend', () => p.remove());
                }
                score++;
                scoreEl.textContent = score;
                cleanup();
            };

            hitbox.addEventListener('touchstart', popHandler, { passive: false });
            hitbox.addEventListener('pointerdown', popHandler);

            container.addEventListener('animationend', cleanup);

            viewport.appendChild(container);

            void container.offsetHeight;
            container.style.animation = `floatUp ${duration}s linear forwards`;

            setTimeout(cleanup, duration * 1000 + 200);

            if (isPaused) container.classList.add('paused');
        }

        function scheduleSpawn() {
            if (isPaused) return;

            if (activeBalloons.size >= MAX_ACTIVE) {
                spawnTimeout = setTimeout(scheduleSpawn, 200);
                return;
            }

            const left = pickRandomX();
            createBalloon(left);

            let delay;
            if (isWaveMode) {
                delay = 150 + Math.random() * 100;
            } else {
                delay = 600 + Math.random() * 200;
            }

            spawnTimeout = setTimeout(scheduleSpawn, delay);
        }

        function resetField() {
            activeBalloons.clear();
            prevX = -1;
            document.querySelectorAll('.balloon-container').forEach(b => b.remove());
            score = 0;
            scoreEl.textContent = score;
        }

        function toggleWaveMode() {
            isWaveMode = !isWaveMode;
            resetField();

            if (isWaveMode) waveBtn.classList.add('active-mode');
            else waveBtn.classList.remove('active-mode');

            if (isPaused) {
                modeChangedOnPause = true;
            } else {
                clearTimeout(spawnTimeout);
                scheduleSpawn();
            }
        }

        let waveBusy = false;

        function waveTap(e) {
            e.preventDefault();
            e.stopPropagation();

            if (waveBusy) return;
            waveBusy = true;
            setTimeout(() => { waveBusy = false; }, 350);

            unlockAudio();
            toggleWaveMode();
        }

        waveBtn.addEventListener('touchstart', waveTap, { passive: false });
        waveBtn.addEventListener('pointerdown', waveTap);

        let pauseBusy = false;

        function pauseTap(e) {
            e.preventDefault();
            e.stopPropagation();

            if (pauseBusy) return;
            pauseBusy = true;
            setTimeout(() => { pauseBusy = false; }, 350);

            unlockAudio();
            togglePause();
        }

        pauseBtn.addEventListener('touchstart', pauseTap, { passive: false });
        pauseBtn.addEventListener('pointerdown', pauseTap);

        function togglePause() {
            isPaused = !isPaused;

            document.querySelectorAll('.balloon-container').forEach(b => {
                if (isPaused) {
                    b.classList.add('paused');
                } else {
                    b.classList.add('paused');
                    void b.offsetHeight;
                    b.classList.remove('paused');
                }
            });

            if (isPaused) {
                pauseBtn.style.opacity = '1';
                clearTimeout(spawnTimeout);
            } else {
                pauseBtn.style.opacity = '0.4';
                clearTimeout(spawnTimeout);
                if (modeChangedOnPause) {
                    modeChangedOnPause = false;
                    scheduleSpawn();
                } else {
                    spawnTimeout = setTimeout(scheduleSpawn, 300);
                }
            }
        }

        scheduleSpawn();
